// 回文の半径を返すのに注意
vector<int> manacher(string s){
  vector<int> res(s.size());
  int i=0,j=0;
  while(i<s.size()){
    while(i-j>=0 && i+j<s.size() && s[i-j] == s[i+j]) j++;
    res[i]=j;
    int k=1;
    while(i-k>=0 && k+res[i-k]<j){
      res[i+k]=res[i-k];
      k++;
    }
    i+=k,j-=k;
  }
  return res;
}
