ll mod=998244353;
int MAX=1e6;
vector<ll> fac(MAX),inv(MAX),finv(MAX);
void COMinit(){
  fac[0]=fac[1]=1;
  inv[1]=1;
  finv[0]=finv[1]=1;
  rep(i,2,MAX){
    fac[i]=fac[i-1]*i%mod;
    inv[i]=mod-(mod/i)*inv[mod%i]%mod;
    finv[i]=finv[i-1]*inv[i]%mod;
  }
  return;
}

ll COM(ll n,ll k){
  if(n<k || n<0 || k<0) return 0;
  return (fac[n]*finv[k]%mod)*finv[n-k]%mod;
}
